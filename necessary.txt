# ========== Timestamped-only mix from filtered/ shards (ChronoGPT-style) ==========
# Requires real parseable dates ≤ 2011-12-31. No trusted-without-date.
# Excludes FineWeb by default. Optional hard year text scan.

# Prefer filtered/ (or filtered_edu/ if that is what you have).
# Expected stems: wikipedia_2012 news_2012 usenet pes2o  (+ qa_2012 from qa/)

# 1) Sample one source first (optional)
python -m train.data.build_timestamped_mix \
  --filtered-dir $SN38_DATA_ROOT/filtered \
  --cutoff-year 2011 \
  --year-mode hard \
  --sources wikipedia_2012 \
  --max-docs-per-source 100000 \
  --out-dir $SN38_DATA_ROOT/filtered_timestamped_2011_sample \
  --shards-only

# 2) Full: write timestamped shards + mix
python -m train.data.build_timestamped_mix \
  --filtered-dir $SN38_DATA_ROOT/filtered \
  --cutoff-year 2011 \
  --year-mode hard \
  --out-dir $SN38_DATA_ROOT/filtered_timestamped_2011 \
  --out-mix $SN38_DATA_ROOT/mix/train_mix_timestamped_2011.jsonl \
  --total-tokens 20e9

# Optional: include books (weaker dates)
#   ... --include-books

# 3) Train from scratch on timestamped mix (keep 3B ckpt; new out dir)
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True
export SN38_PEAK_TFLOPS=989
TOK=$SN38_DATA_ROOT/tokenizer
# TOK=$SN38_DATA_ROOT/checkpoints/chrono-2011-a-3-FP32   # if tokenizer only lives there
mkdir -p $SN38_DATA_ROOT/logs $SN38_DATA_ROOT/checkpoints/from_scratch_ts_2011

torchrun --nproc_per_node 2 -m train.train_clm \
  --mix $SN38_DATA_ROOT/mix/train_mix_timestamped_2011.jsonl \
  --arch nanochrono --init-new --tokenizer "$TOK" \
  --total-tokens 20e9 --batch-size 10 --grad-accum 20 --num-workers 4 \
  --schedule wsd --decay-frac 0.2 --bf16-grad-sync \
  --out $SN38_DATA_ROOT/checkpoints/from_scratch_ts_2011 \
  2>&1 | tee $SN38_DATA_ROOT/logs/train_from_scratch_ts_2011.log
